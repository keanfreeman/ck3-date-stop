# "The picked date is in the future, an alarm slot is free and it isn't a duplicate."
dsc_add_sgui = {
	scope = character
	is_valid = {
		OR = {
			var:dsc_py > current_year
			AND = {
				var:dsc_py = current_year
				OR = {
					var:dsc_pm > current_month
					AND = { var:dsc_pm = current_month var:dsc_pd > current_day }
				}
			}
		}
		OR = {
			NOT = { has_variable = dsc_a1 }
			NOT = { has_variable = dsc_a2 }
			NOT = { has_variable = dsc_a3 }
			NOT = { has_variable = dsc_a4 }
			NOT = { has_variable = dsc_a5 }
			NOT = { has_variable = dsc_a6 }
		}
		NOT = { AND = { has_variable = dsc_a1 var:dsc_a1_y = var:dsc_py var:dsc_a1_m = var:dsc_pm var:dsc_a1_d = var:dsc_pd } }
		NOT = { AND = { has_variable = dsc_a2 var:dsc_a2_y = var:dsc_py var:dsc_a2_m = var:dsc_pm var:dsc_a2_d = var:dsc_pd } }
		NOT = { AND = { has_variable = dsc_a3 var:dsc_a3_y = var:dsc_py var:dsc_a3_m = var:dsc_pm var:dsc_a3_d = var:dsc_pd } }
		NOT = { AND = { has_variable = dsc_a4 var:dsc_a4_y = var:dsc_py var:dsc_a4_m = var:dsc_pm var:dsc_a4_d = var:dsc_pd } }
		NOT = { AND = { has_variable = dsc_a5 var:dsc_a5_y = var:dsc_py var:dsc_a5_m = var:dsc_pm var:dsc_a5_d = var:dsc_pd } }
		NOT = { AND = { has_variable = dsc_a6 var:dsc_a6_y = var:dsc_py var:dsc_a6_m = var:dsc_pm var:dsc_a6_d = var:dsc_pd } }
	}
	effect = { dsc_add = yes }
}

dsc_clear_sgui = {
	scope = character
	is_shown = {
		OR = {
			has_variable = dsc_a1
			has_variable = dsc_a2

			has_variable = dsc_a5
			has_variable = dsc_a6
			has_variable = dsc_pr
			has_variable = dsc_prep
		}
	}
	effect = { dsc_clear_all = yes }
}
